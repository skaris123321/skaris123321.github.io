⚠️ ВАЖНО: ОТКУДА БРАТЬ АРТИКУЛ
═══════════════════════════════════════════════════════════════════════════════
Артикул для сайта берется из КОММЕНТАРИЯ к товару в 1С (не из поля "Артикул")!

Пример товара в 1С:
  Наименование: Шкаф АВР 25А
  Артикул: 12345                    ← НЕ ИСПОЛЬЗОВАТЬ
  Комментарий: АВР-2-25-М-РОСЭК     ← ИСПОЛЬЗОВАТЬ ЭТО
  Цена: 45000


ПРИМЕР КОДА
═══════════════════════════════════════════════════════════════════════════════
Функция ОбновитьЦеныНаСайте()
    
    // Получаем товары (артикул из комментария!)
    Запрос = Новый Запрос;
    Запрос.Текст = "ВЫБРАТЬ
    |    Номенклатура.Комментарий КАК article,
    |    Номенклатура.Цена КАК price
    |ИЗ Справочник.Номенклатура КАК Номенклатура
    |ГДЕ Номенклатура.Цена > 0
    |    И Номенклатура.Комментарий <> """"";
    
    Результат = Запрос.Выполнить();
    Выборка = Результат.Выбрать();
    
    // Формируем массив
    МассивЦен = Новый Массив;
    Пока Выборка.Следующий() Цикл
        Товар = Новый Структура;
        Товар.Вставить("article", СокрЛП(Выборка.article));
        Товар.Вставить("price", Выборка.price);
        МассивЦен.Добавить(Товар);
    КонецЦикла;
    
    // Формируем JSON
    Данные = Новый Структура;
    Данные.Вставить("prices", МассивЦен);
    
    ЗаписьJSON = Новый ЗаписьJSON;
    ЗаписьJSON.УстановитьСтроку();
    ЗаписатьJSON(ЗаписьJSON, Данные);
    JSONСтрока = ЗаписьJSON.Закрыть();
    
    // Отправляем
    Соединение = Новый HTTPСоединение("localhost", 3000);
    Запрос = Новый HTTPЗапрос("/api/products/update-prices");
    Запрос.Заголовки.Вставить("Content-Type", "application/json");
    Запрос.Заголовки.Вставить("X-API-Key", "ваш-ключ");
    Запрос.УстановитьТелоИзСтроки(JSONСтрока);
    
    Ответ = Соединение.ОтправитьДляОбработки(Запрос);
    
    Возврат Ответ.КодСостояния = 200;
    
КонецФункции
